## server.js

```javascript
import { createServer } from 'node:http';
import { readFile, writeFile, mkdir, stat } from 'node:fs/promises';
import { randomBytes, scryptSync, timingSafeEqual } from 'node:crypto';
import { extname, join, normalize } from 'node:path';

const root=process.cwd(),dataDir=join(root,'data'),dbFile=join(dataDir,'travel-go-db.json'),sessions=new Map();
const mime={'.html':'text/html; charset=utf-8','.js':'text/javascript; charset=utf-8','.css':'text/css; charset=utf-8','.json':'application/json; charset=utf-8'};
async function database(){try{return JSON.parse(await readFile(dbFile,'utf8'))}catch{await mkdir(dataDir,{recursive:true});const initial={users:[],bookings:[]};await writeFile(dbFile,JSON.stringify(initial,null,2));return initial}}
const save=db=>writeFile(dbFile,JSON.stringify(db,null,2));
const send=(res,code,value)=>{res.writeHead(code,{'Content-Type':'application/json; charset=utf-8'});res.end(JSON.stringify(value))};
function hash(password,salt=randomBytes(16).toString('hex')){return{salt,value:scryptSync(password,salt,64).toString('hex')}}
function validPassword(password,user){const value=Buffer.from(hash(password,user.salt).value,'hex'),saved=Buffer.from(user.passwordHash,'hex');return value.length===saved.length&&timingSafeEqual(value,saved)}
function createSession(user){const token=randomBytes(32).toString('hex');sessions.set(token,{userId:user.id,expires:Date.now()+604800000});return token}
function safeUser(user){return{id:user.id,name:user.name,email:user.email}}
function currentUser(req,db){const token=(req.headers.authorization||'').replace('Bearer ','');const session=sessions.get(token);return session&&session.expires>Date.now()?db.users.find(u=>u.id===session.userId):null}
async function body(req){let raw='';for await(const part of req)raw+=part;try{return JSON.parse(raw||'{}')}catch{throw new Error('Invalid request data.')}}
createServer(async(req,res)=>{try{const url=new URL(req.url,`http://${req.headers.host}`);if(req.method==='POST'&&url.pathname==='/api/auth/register'){const{name,email,password}=await body(req);if(!name||!email||!password||password.length<6)return send(res,400,{error:'Name, email and a 6-character password are required.'});const db=await database();if(db.users.some(u=>u.email===email.toLowerCase()))return send(res,409,{error:'An account already exists for this email.'});const secured=hash(password),user={id:randomBytes(12).toString('hex'),name:name.trim(),email:email.toLowerCase(),salt:secured.salt,passwordHash:secured.value,createdAt:new Date().toISOString()};db.users.push(user);await save(db);return send(res,201,{user:safeUser(user),token:createSession(user)})}if(req.method==='POST'&&url.pathname==='/api/auth/login'){const{email,password}=await body(req),db=await database(),user=db.users.find(u=>u.email===String(email).toLowerCase());if(!user||!validPassword(password||'',user))return send(res,401,{error:'Email or password is incorrect.'});return send(res,200,{user:safeUser(user),token:createSession(user)})}if(req.method==='POST'&&url.pathname==='/api/bookings'){const db=await database(),user=currentUser(req,db);if(!user)return send(res,401,{error:'Please login before making a booking request.'});const data=await body(req);if(!data.tripId||!data.phone||!data.travelMonth)return send(res,400,{error:'Please complete the booking form.'});const booking={id:randomBytes(10).toString('hex'),userId:user.id,tripId:data.tripId,tripName:data.tripName,phone:data.phone,travelMonth:data.travelMonth,travellers:data.travellers,notes:data.notes||'',status:'requested',createdAt:new Date().toISOString()};db.bookings.push(booking);await save(db);return send(res,201,{booking})}if(req.method==='GET'&&url.pathname==='/api/health')return send(res,200,{status:'ok'});const requested=url.pathname==='/'?'/standalone.html':url.pathname,file=normalize(join(root,requested));if(!file.startsWith(root))return send(res,403,{error:'Forbidden'});await stat(file);res.writeHead(200,{'Content-Type':mime[extname(file)]||'application/octet-stream'});res.end(await readFile(file))}catch(error){if(error.code==='ENOENT')return send(res,404,{error:'Not found'});console.error(error);send(res,500,{error:'Server error'})}}).listen(3000,()=>console.log('Travel Go is running at http://localhost:3000'));

'Seat preference<select name="vehicle"><option value="AC Sleeper">AC Sleeper</option><option value="AC Seater">AC Seater</option><option value="Volvo">Volvo</option></select>' : 'Cab type<select name="vehicle"><option value="Hatchback">Hatchback</option><option value="Sedan">Sedan</option><option value="SUV">SUV</option><option value="Tempo Traveller">Tempo Traveller</option></select>';

document.querySelector('#transport-form').onsubmit = async event => {
  event.preventDefault();
  const fields = Object.fromEntries(new FormData(event.target));
  const whatsappWindow = window.open('', '_blank');
  try {
    await request('/api/bookings', { method:'POST', body:JSON.stringify({ tripId:`${selectedService}-${Date.now()}`, tripName:`${selectedService === 'bus' ? 'Bus' : 'Cab'}: ${fields.from} to ${fields.to}`, phone:fields.phone, travelMonth:fields.journeyDate, travellers:fields.passengers, notes:`Vehicle: ${fields.vehicle}. ${fields.notes || ''}` }) });
    const title = selectedService === 'bus' ? 'Bus booking' : 'Cab booking';
    sendBookingToOwner({type:title,phone:fields.phone,body:`*Route:* ${fields.from} → ${fields.to}\n*Journey date:* ${fields.journeyDate}\n*Travellers:* ${fields.passengers}\n*Vehicle:* ${fields.vehicle}\n*Notes:* ${fields.notes || 'None'}`}, whatsappWindow);
    transportDialog.close(); event.target.reset(); notify(`${selectedService === 'bus' ? 'Bus' : 'Cab'} request saved. Opening WhatsApp with the full request…`);
  } catch (error) { if(whatsappWindow&&!whatsappWindow.closed)whatsappWindow.close(); document.querySelector('#transport-error').textContent = error.message; }
};

function updateRideEstimate() {
  const passengers = document.querySelector('#transport-form [name="passengers"]').value;
  const passengerCount = passengers === '5+' ? 5 : Number(passengers);
  const vehicle = document.querySelector('#transport-form [name="vehicle"]').value;
  const vehicleExtra = vehicle === 'SUV' || vehicle === 'Volvo' ? 450 : vehicle === 'Tempo Traveller' ? 1100 : vehicle === 'Sedan' || vehicle === 'AC Sleeper' ? 200 : 0;
  const base = selectedService === 'bus' ? 650 : 1200;
  document.querySelector('#price-estimate').textContent = `Estimated starting fare: ₹${base + vehicleExtra + Math.max(0, passengerCount - 1) * 120} · final fare depends on route and availability`;
}
document.querySelector('#transport-form').addEventListener('change', updateRideEstimate);
